# Cadence — everything the app needs from outside Python itself.
#
# Install with:
#     pip install -r requirements.txt
#
# The "==" pins an exact version. That means the app will behave
# the same on your machine next year as it does today, instead of
# silently breaking when one of these releases an update.

Flask==3.1.3        # the web framework: turns Python functions into web pages
waitress==3.0.2     # the production server that actually faces the internet

# Deliberately NOT here:
#
#   sqlite3   - built into Python, no install needed
#   zoneinfo  - built in (Python 3.9+), handles West Africa Time
#   secrets   - built in, generates the secret key
#
# Cadence needs exactly two outside packages. That is on purpose:
# every dependency is something that can break, need updating, or
# turn into a security hole.
#
# Requires Python 3.9 or newer (3.13 is what it was built on).
